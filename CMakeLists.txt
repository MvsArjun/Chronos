cmake_minimum_required(VERSION 3.24)
project(Chronos VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CHRONOS_TESTS "Build GoogleTest tests" ON)
option(CHRONOS_BENCHMARKS "Build Google Benchmark suite" ON)
option(CHRONOS_PYTHON "Build pybind11 extension" ON)
option(CHRONOS_CRC_BASELINE "Use bitwise CRC for benchmark reproduction" OFF)
set(CHRONOS_SANITIZER "none" CACHE STRING "none, address-undefined, or thread")
if(NOT CHRONOS_SANITIZER MATCHES "^(none|address-undefined|thread)$")
  message(FATAL_ERROR "Invalid CHRONOS_SANITIZER")
endif()
if(NOT CHRONOS_SANITIZER STREQUAL "none")
  if(WIN32 OR MSVC)
    message(FATAL_ERROR "Chronos sanitizer configurations require Linux GCC/Clang")
  endif()
  if(CHRONOS_SANITIZER STREQUAL "thread")
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=thread)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
find_package(Threads REQUIRED)
add_library(chronos src/market_data/data.cpp)
set_target_properties(chronos PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(chronos PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(chronos PUBLIC Threads::Threads)
if(CHRONOS_CRC_BASELINE)
  target_compile_definitions(chronos PRIVATE CHRONOS_CRC_BITWISE_BASELINE=1)
endif()
if(MSVC)
  target_compile_options(chronos PUBLIC /W4 /permissive-)
else()
  target_compile_options(chronos PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(chronos_cli src/main.cpp)
target_link_libraries(chronos_cli PRIVATE chronos)
include(FetchContent)
if(CHRONOS_TESTS)
  enable_testing()
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.16.0.zip URL_HASH SHA256=A9607C9215866BD425A725610C5E0F739EEB50887A57903DF48891446CE6FB3C DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  file(GLOB CHRONOS_TEST_SOURCES CONFIGURE_DEPENDS tests/*_tests.cpp)
  add_executable(chronos_tests ${CHRONOS_TEST_SOURCES})
  target_link_libraries(chronos_tests PRIVATE chronos GTest::gtest_main)
  add_test(NAME chronos_tests COMMAND chronos_tests)
endif()
if(CHRONOS_BENCHMARKS)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(googlebenchmark URL https://github.com/google/benchmark/archive/refs/tags/v1.9.2.zip URL_HASH SHA256=A13CDCD3B0E3725E371F977A2855FECD651FC0236C67DA16E325B726057B15B4 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googlebenchmark)
  add_executable(chronos_bench benchmarks/bench.cpp)
  target_link_libraries(chronos_bench PRIVATE chronos benchmark::benchmark_main)
  add_executable(chronos_latency benchmarks/latency.cpp)
  target_link_libraries(chronos_latency PRIVATE chronos)
endif()
if(CHRONOS_PYTHON)
  find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND11_CMAKE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  list(APPEND CMAKE_PREFIX_PATH "${PYBIND11_CMAKE_DIR}")
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core python/bindings/module.cpp)
  target_link_libraries(_core PRIVATE chronos)
  if(NOT SKBUILD)
    set_target_properties(_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/python/chronos")
  endif()
  install(TARGETS _core LIBRARY DESTINATION chronos)
endif()
install(TARGETS chronos chronos_cli)
install(DIRECTORY include/ DESTINATION include)
